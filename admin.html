/************************************************************ * ARQSELECT - BACKEND ADMINISTRATIVO * Google Apps Script - Code.gs ************************************************************/ const CONFIG = { // ========================================================= // LOGIN ADMINISTRATIVO // ========================================================= ADMIN_USUARIO: "alefesantos27", ADMIN_SENHA: "@Sucesso01", // ========================================================= // PLANILHA // ========================================================= ID_DA_PLANILHA: "COLOQUE_AQUI_O_ID_DA_SUA_PLANILHA", NOME_DA_ABA: "Respostas ao formulário 1", // Tempo da sessão em minutos SESSAO_MINUTOS: 120 }; // ============================================================ // CACHE DE SESSÕES // ============================================================ const CACHE_PREFIX = "ARQSELECT_SESSION_"; // ============================================================ // GET // ============================================================ function doGet(e) { return respostaJSON({ sucesso: true, sistema: "ARQSELECT", mensagem: "API ARQSELECT funcionando.", horario: new Date().toISOString() }); } // ============================================================ // POST // ============================================================ function doPost(e) { try { const dados = obterParametros(e); const acao = String( dados.acao || "" ).trim(); switch (acao) { case "login_admin": return loginAdmin(dados); case "validar_sessao_admin": return validarSessaoAdmin(dados); case "logout_admin": return logoutAdmin(dados); case "admin_teste": exigirSessao(dados.token); return respostaJSON({ sucesso: true, autorizado: true, mensagem: "API administrativa funcionando corretamente." }); case "projetos": exigirSessao(dados.token); return obterProjetos(); case "projeto": exigirSessao(dados.token); return obterProjeto( dados.id ); case "atualizarStatus": exigirSessao(dados.token); return atualizarStatus( dados.linha, dados.status ); default: return respostaJSON({ sucesso: false, autorizado: false, mensagem: "Ação não reconhecida." }); } } catch (erro) { return respostaJSON({ sucesso: false, autorizado: false, mensagem: erro && erro.message ? erro.message : "Erro interno do servidor." }); } } // ============================================================ // PARÂMETROS // ============================================================ function obterParametros(e) { if ( e && e.parameter ) { return e.parameter; } return {}; } // ============================================================ // LOGIN // ============================================================ function loginAdmin(dados) { const usuario = String( dados.usuario || "" ).trim(); const senha = String( dados.senha || "" ); if ( usuario !== CONFIG.ADMIN_USUARIO || senha !== CONFIG.ADMIN_SENHA ) { return respostaJSON({ sucesso: false, autorizado: false, mensagem: "Usuário ou senha inválidos." }); } const token = criarSessao( usuario ); return respostaJSON({ sucesso: true, autorizado: true, token: token, usuario: usuario, mensagem: "Login realizado com sucesso." }); } // ============================================================ // CRIAR SESSÃO // ============================================================ function criarSessao(usuario) { const token = Utilities.getUuid() + "-" + Utilities.getUuid(); const dados = { usuario: usuario, criadoEm: new Date().getTime(), expiraEm: new Date().getTime() + ( CONFIG.SESSAO_MINUTOS * 60 * 1000 ) }; CacheService .getScriptCache() .put( CACHE_PREFIX + token, JSON.stringify(dados), CONFIG.SESSAO_MINUTOS * 60 ); return token; } // ============================================================ // VALIDAR SESSÃO // ============================================================ function validarSessaoAdmin(dados) { const sessao = obterSessao( dados.token ); if (!sessao) { return respostaJSON({ sucesso: false, autorizado: false, mensagem: "Sessão expirada." }); } return respostaJSON({ sucesso: true, autorizado: true, usuario: sessao.usuario, mensagem: "Sessão válida." }); } // ============================================================ // OBTER SESSÃO // ============================================================ function obterSessao(token) { if (!token) { return null; } const cache = CacheService .getScriptCache(); const texto = cache.get( CACHE_PREFIX + token ); if (!texto) { return null; } try { const sessao = JSON.parse(texto); if ( new Date().getTime() > Number( sessao.expiraEm ) ) { cache.remove( CACHE_PREFIX + token ); return null; } return sessao; } catch (erro) { return null; } } // ============================================================ // EXIGIR SESSÃO // ============================================================ function exigirSessao(token) { const sessao = obterSessao(token); if (!sessao) { throw new Error( "Acesso não autorizado. Faça login novamente." ); } return sessao; } // ============================================================ // LOGOUT // ============================================================ function logoutAdmin(dados) { if (dados.token) { CacheService .getScriptCache() .remove( CACHE_PREFIX + dados.token ); } return respostaJSON({ sucesso: true, autorizado: false, mensagem: "Sessão encerrada." }); } // ============================================================ // PLANILHA // ============================================================ function obterPlanilha() { if ( !CONFIG.ID_DA_PLANILHA || CONFIG.ID_DA_PLANILHA === "COLOQUE_AQUI_O_ID_DA_SUA_PLANILHA" ) { throw new Error( "Configure o ID_DA_PLANILHA no Code.gs." ); } return SpreadsheetApp .openById( CONFIG.ID_DA_PLANILHA ); } // ============================================================ // ABA // ============================================================ function obterAba() { const planilha = obterPlanilha(); let aba = planilha.getSheetByName( CONFIG.NOME_DA_ABA ); if (!aba) { aba = planilha.getSheets()[0]; } if (!aba) { throw new Error( "Nenhuma aba encontrada na planilha." ); } return aba; } // ============================================================ // LER PLANILHA // ============================================================ function lerPlanilha() { const aba = obterAba(); const ultimaLinha = aba.getLastRow(); const ultimaColuna = aba.getLastColumn(); if ( ultimaLinha < 1 || ultimaColuna < 1 ) { return []; } const valores = aba.getRange( 1, 1, ultimaLinha, ultimaColuna ).getDisplayValues(); if ( !valores.length ) { return []; } const cabecalhos = valores[0].map( valor => String( valor || "" ).trim() ); const registros = []; for ( let i = 1; i < valores.length; i++ ) { const linha = valores[i]; const objeto = { _linha: i + 1 }; let possuiDados = false; for ( let c = 0; c < cabecalhos.length; c++ ) { const chave = cabecalhos[c]; if (!chave) { continue; } const valor = linha[c] !== undefined ? linha[c] : ""; objeto[chave] = valor; if ( String( valor || "" ).trim() !== "" ) { possuiDados = true; } } if (possuiDados) { registros.push( objeto ); } } return registros; } // ============================================================ // PROJETOS // ============================================================ function obterProjetos() { const projetos = lerPlanilha(); return respostaJSON({ sucesso: true, autorizado: true, projetos: projetos, total: projetos.length }); } // ============================================================ // PROJETO INDIVIDUAL // ============================================================ function obterProjeto(id) { const projetos = lerPlanilha(); const procurado = String( id || "" ).trim(); const projeto = projetos.find( item => { const valor = String( item["ID PROJETO"] || item["NUMERO DO PROJETO"] || "" ).trim(); return valor === procurado; } ); if (!projeto) { return respostaJSON({ sucesso: false, autorizado: true, mensagem: "Projeto não encontrado." }); } return respostaJSON({ sucesso: true, autorizado: true, projeto: projeto }); } // ============================================================ // ATUALIZAR STATUS // ============================================================ function atualizarStatus( linha, status ) { const permitidos = [ "Novo", "Em análise", "Orçamento", "Fechado", "Cancelado" ]; const novoStatus = String( status || "" ).trim(); if ( !permitidos.includes( novoStatus ) ) { return respostaJSON({ sucesso: false, autorizado: true, mensagem: "Status inválido." }); } const numeroLinha = Number(linha); if ( !Number.isInteger( numeroLinha ) || numeroLinha < 2 ) { return respostaJSON({ sucesso: false, autorizado: true, mensagem: "Linha inválida." }); } const aba = obterAba(); const ultimaColuna = aba.getLastColumn(); const cabecalhos = aba .getRange( 1, 1, 1, ultimaColuna ) .getDisplayValues()[0]; let colunaStatus = -1; for ( let i = 0; i < cabecalhos.length; i++ ) { const nome = String( cabecalhos[i] || "" ) .trim() .toUpperCase(); if ( nome === "STATUS" ) { colunaStatus = i + 1; break; } } if ( colunaStatus === -1 ) { throw new Error( "A coluna STATUS não foi encontrada na planilha." ); } aba .getRange( numeroLinha, colunaStatus ) .setValue( novoStatus ); SpreadsheetApp .flush(); return respostaJSON({ sucesso: true, autorizado: true, status: novoStatus, linha: numeroLinha, mensagem: "Status atualizado com sucesso." }); } // ============================================================ // JSON // ============================================================ function respostaJSON(objeto) { return ContentService .createTextOutput( JSON.stringify( objeto ) ) .setMimeType( ContentService .MimeType .JSON ); }
